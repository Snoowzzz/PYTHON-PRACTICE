import time
for i in range(5):
    print(i, end=" ", flush=True)
    time.sleep(0.5)  # Simulates a delay

##some important info 
Without flush=True:  [ 0, 1, 2, 3, 4 ]  --> 
  (Held in buffer, appears all at once at the end)
With flush=True:     0 --> 1 --> 2 -->   -->
  (Pushed to the screen one by one instantly)

# EXPLANATION: for x in y syntax
# When you write: for pos in word
# Python iterates through each CHARACTER in the word string one by one
# 'pos' is not a position/index, it's the ACTUAL CHARACTER at that position
# 
# Example:
# If word = "Hi"
# Iteration 1: pos = "H" (the first character)
# Iteration 2: pos = "i" (the second character)
#
# Another example with for i in range(5):
# range(5) generates: 0, 1, 2, 3, 4
# So i takes these values one by one
# 
# KEY DIFFERENCE:
# for i in range(5)      -> i is a NUMBER (0, 1, 2, 3, 4)
# for pos in "Hello"     -> pos is a CHARACTER ("H", "e", "l", "l", "o")
#
# YOUR CODE ISSUE:

# This causes an error because you can't do "H" + 1
